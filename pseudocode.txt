FUNCTION baca_angka (min : INTEGER, max : INTEGER) : INTEGER
    DECLARE angka : INTEGER
    INPUT angka

    WHILE (INPUT NOT VALID) OR (angka < min) OR (angka > max) DO 
        OUTPUT "Input salah! Masukkan angka ", min, " - ", max, " : "
        INPUT angka
    END WHILE 
    RETURN angka
END FUNCTION

FUNCTION baca_desimal (min : DOUBLE , max : DOUBLE ) : DOUBLE
    DECLARE angka : DOUBLE
    INPUT angka

    WHILE (INPUT NOT VALID) OR (angka < min) OR (angka > max) DO 
        OUTPUT "Input salah! Masukkan angka ", min, " - ", max, " : "
        INPUT angka
    END WHILE
    RETURN angka
END FUNCTION

FUNCTION nama_kategori (pilihan : INTEGER) : STRING 
    IF (pilihan == 1) THEN RETURN "buku"
    IF (pilihan == 2) THEN RETURN "catatan"
    IF (pilihan == 3) THEN RETURN "tutorial"
    IF (pilihan == 4) THEN RETURN "video_pembelajaran"
    RETURN "online_course"
END FUNCTION

PROCEDURE input (IN/OUT belajar : data_sumber, n : INTEGER) 
    DECLARE pilih : INTEGER

    SET belajar.id = n + 1

    OUTPUT "===== Menambah data sumber belajar ====="
    OUTPUT "Judul (1 kata, pakai _ untuk spasi) : "
    INPUT belajar.judul

    CALL menu_kategori()
    SET pilih = baca_angka(1, 5)
    SET belajar.kategori = nama_kategori(pilih)

    CALL menu_tingkat()
    SET pilih = baca_angka(1, 3)
    IF (pilih == 1) THEN
        SET belajar.tingkat_kesulitan = "mudah"
    ELSE IF (pilih == 2) THEN
        SET belajar.tingkat_kesulitan = "sedang"
    ELSE
        SET belajar.tingkat_kesulitan = "sulit"
    END IF

    OUTPUT "Rating (1-5) = "
    SET belajar.rating = baca_desimal(1, 5)

    OUTPUT "Durasi belajar (menit) = "
    SET belajar.waktu = baca_angka(1, INT_MAX)

    OUTPUT "Data berhasil ditambahkan (ID ", belajar.id, ")."
END PROCEDURE

PROCEDURE view_baris (belajar : data_sumber) 
    OUTPUT belajar.id, belajar.judul, belajar.kategori, belajar.tingkat_kesulitan, belajar.rating, belajar.waktu
END PROCEDURE

PROCEDURE view_arr (belajar[] : data_sumber, n : INTEGER) 
    IF (n == 0) THEN
        OUTPUT "(tidak ada data)"
    END IF
    FOR  i = 0 TO n-1 DO 
        CALL view_baris(belajar[i])
    END FOR
END PROCEDURE

PROCEDURE view_terkelompokan (tingkat : STRING, belajar[] : data_sumber, n : INTEGER) 
    DECLARE jumlah : INTEGER
    SET jumlah = 0

    FOR i = 0 TO n-1 DO 
        IF (belajar[i].tingkat_kesulitan == tingkat) THEN
            CALL view_baris(belajar[i])
            SET jumlah = jumlah + 1
        END IF
    END FOR
    IF (jumlah == 0) THEN
        OUTPUT "(tidak ada data)"
    END IF
END PROCEDURE


========================== load_save_file ========================================
PROCEDURE load (IN/OUT belajar[] : data_sumber, IN/OUT n : INTEGER) 
    OPEN FILE "data.txt" FOR READING AS out
    IF (NOT out) THEN 
        RETURN
    END IF
    WHILE (READ out INTO belajar[n].id, belajar[n].judul, belajar[n].kategori, belajar[n].tingkat_kesulitan, belajar[n].rating, belajar[n].waktu) DO 
        SET n = n + 1
    END WHILE
    CLOSE FILE out
END PROCEDURE

PROCEDURE save (belajar : data_sumber) 
    OPEN FILE "data.txt" FOR APPEND AS in

    IF (in IS OPEN) THEN 
        WRITE in FROM belajar.id, belajar.judul, belajar.kategori, belajar.tingkat_kesulitan, belajar.rating, belajar.waktu 
    ELSE
        OUTPUT "File data.txt tidak ditemukan."
    END IF
    CLOSE FILE in
END PROCEDURE

================================ calculation.hpp =========================================
FUNCTION totaltime (belajar[] : data_sumber, jumlah : INTEGER) : DOUBLE
    IF (jumlah <= 0) THEN
        RETURN 0
    END IF

    RETURN belajar[jumlah-1].waktu + totaltime(belajar, jumlah-1)
END FUNCTION

FUNCTION mean(belajar[] : data_sumber, jumlah : INTEGER, properties : STRING) : DOUBLE
    IF (jumlah == 0) 
        THEN RETURN 0
    END IF
    DECLARE sum, mean : DOUBLE
    SET sum = 0
    SET mean = 0
    
        IF (properties == "rating") THEN 
            FOR i = 0 TO jumlah-1 DO 
                SET sum = sum + belajar[i].rating
            END FOR
        ELSE IF (properties == "waktu") THEN
            SET sum = totaltime(belajar, jumlah)
        END IF
    
    SET mean = sum/jumlah
    RETURN mean
END FUNCTION

FUNCTION findmax (belajar[] : data_sumber, jumlah : INTEGER, properties : STRING) : DOUBLE
    DECLARE max : DOUBLE
    SET max = 0

    FOR i = 0 TO jumlah-1 DO
        IF (properties == "rating") THEN
            IF (max < belajar[i].rating) THEN
                SET max = belajar[i].rating
            END IF
        ELSE IF (properties == "waktu") THEN
            IF (max < belajar[i].waktu) THEN
                SET max = belajar[i].waktu
            END IF
        END IF
    END FOR

    RETURN max
END FUNCTION


FUNTION findmin (belajar[] : data_sumber, jumlah : INTEGER, properties : STRING) : DOUBLE
    DECLARE min : DOUBLE
    SET min = INT_MAX

    FOR i = 0 TO jumlah-1 DO
        IF (properties == "rating") THEN
            IF (min > belajar[i].rating) THEN
                SET min = belajar[i].rating
            END IF
        ELSE IF (properties == "waktu") THEN
            IF (min > belajar[i].waktu) THEN
                SET min = belajar[i].waktu
            END IF
        END IF
    END FOR

    RETURN min
END FUNCTION

PROCEDURE bubble_ascending(IN/OUT belajar[] : data_sumber, jumlah : INTEGER, properties : STRING)
    FOR i = 0 TO jumlah-1 DO
        FOR j = i+1 TO jumlah-1 DO
            IF (properties == "rating") THEN
                IF (belajar[i].rating > belajar[j].rating) THEN
                    DECLARE temp : data_sumber
                    SET temp = belajar[j]
                    SET belajar[j] = belajar[i]
                    SET belajar[i] = temp
                END IF
            END IF

            IF (properties == "waktu") THEN
                IF (belajar[i].waktu > belajar[j].waktu) THEN
                    DECLARE temp : data_sumber
                    SET temp = belajar[j]
                    SET belajar[j] = belajar[i]
                    SET belajar[i] = temp
                END IF
            END IF
        END FOR
    END FOR
END PROCEDURE

PROCEDURE bubble_descending(IN/OUT belajar[] : data_sumber, jumlah : INTEGER, properties : STRING)
    FOR i = 0 TO jumlah-1 DO
        FOR j = i+1 TO jumlah-1 DO

END PROCEDURE

PROCEDURE search_int (belajar[] : data_sumber, n : INTEGER, properties : STRING, target : INTEGER) 
    DECLARE jumlah : INTEGER
    SET jumlah = 0
    FOR i = 0 TO n-1 DO
        IF (properties == "id") THEN
            IF (belajar[i].id == target) THEN
                CALL view_baris(belajar[i])  
                SET jumlah = jumlah + 1
            END IF
        END IF

        IF (properties == "waktu") THEN
            IF (belajar[i].waktu == target) THEN
                CALL view_baris(belajar[i])  
                SET jumlah = jumlah + 1
            END IF
        END IF
    END FOR            
    IF (jumlah == 0) THEN
        OUTPUT "(data tidak ditemukan)"
    END IF
END PROCEDURE

PROCEDURE urut_rating (belajar[] : data_sumber, n : INTEGER) 
    DECLARE pilihan : INTEGER
    
    REPEAT
        CALL menu_urut()
        SET pilihan = baca_angka(1, 3)
        CLEAR SCREEN
        CASE pilihan OF 
            1 : 
                OUTPUT "DATA SUMBER BELAJAR TERURUT RATING (ASCENDING)"
                CALL header()
                CALL bubble_ascending(belajar, n, "rating")
                CALL view_arr(belajar, n)
                OUTPUT "enter untuk lanjut.."
                WAIT FOR KEYPRESS
                CLEAR SCREEN

            2 : 
                OUTPUT "DATA SUMBER BELAJAR TERURUT RATING (DESCENDING)"
                CALL header()
                CALL bubble_descending(belajar, n, "rating")
                CALL view_arr(belajar, n)
                OUTPUT "enter untuk lanjut.."
                WAIT FOR KEYPRESS
                CLEAR SCREEN

            3 : 
                CLEAR SCREEN

            OTHERWISE : 
                CLEAR INPUT BUFFER
        END CASE    
    UNTIL (pilihan == 3)
END PROCEDURE

PROCEDURE menampilkan_data_sumber (belajar[] : data_sumber, n : INTEGER) 
    DECLARE pilihan : INTEGER
    DECLARE jumlah : INTEGER
    DECLARE max_rating, min_rating : DOUBLE
    DECLARE max_waktu, min_waktu : INTEGER
    
    REPEAT
        OUTPUT "rata-rata rating       = "
        OUTPUT mean(belajar, n, "rating")
        OUTPUT "rata-rata durasi waktu = "
        OUTPUT mean(belajar, n, "waktu"), " menit"
        CALL menu_menampilkanData()
        SET pilihan = baca_angka(1, 4)
        CLEAR SCREEN
        
        CASE pilihan OF
            1 :
                CALL tampilkan_seluruh_data(belajar, n)

            2 :
                OUTPUT "DATA SUMBER BELAJAR DENGAN RATING TERTINGGI"
                CALL header()
                SET max_rating = findmax(belajar, n, "rating") 
                CALL search_rating(belajar, n, max_rating)
                OUTPUT "DATA SUMBER BELAJAR DENGAN RATING TERENDAH"
                CALL header()
                SET min_rating = findmin(belajar, n, "rating") 
                CALL search_rating(belajar, n, min_rating)
                OUTPUT "enter untuk lanjut.."
                WAIT FOR KEYPRESS
                CLEAR SCREEN

            3 :
                OUTPUT "DATA SUMBER BELAJAR DENGAN DURASI WAKTU TERPANJANG"
                CALL header()
                SET max_waktu = findmax(belajar, n, "waktu") 
                CALL search_int(belajar, n, "waktu", max_waktu)
                OUTPUT "DATA SUMBER BELAJAR DENGAN DURASI WAKTU TERPENDEK"
                CALL header()
                SET min_waktu = findmin(belajar, n, "waktu") 
                CALL search_int(belajar, n, "waktu", min_waktu)
                OUTPUT "enter untuk lanjut.."
                WAIT FOR KEYPRESS
                CLEAR SCREEN

            4 :
                CLEAR SCREEN

            OTHERWISE : 
                CLEAR INPUT BUFFER
        END CASE
    UNTIL (pilihan == 4)
END PROCEDURE

PROCEDURE cari_data (belajar[] : data_sumber, n : INTEGER) 
    DECLARE pilihan : INTEGER
    DECLARE jumlah : INTEGER
    DECLARE judul, kategori : STRING
    
    REPEAT
        CALL menu_cariData()
        SET pilihan = baca_angka(1, 5)
        CLEAR SCREEN
        
        CASE pilihan OF
            1 :
                DECLARE id : INTEGER
                OUTPUT "ID : "
                SET id = baca_angka(1, INT_MAX)
                CALL header()
                CALL search_int(belajar, n, "id", id)
                OUTPUT "enter untuk lanjut.."
                WAIT FOR KEYPRESS
                CLEAR SCREEN

            2 :
                OUTPUT "Judul (1 kata, pakai _ untuk spasi) : "
                INPUT judul
                CALL header()
                CALL search_string(belajar, n, "judul", judul)
                OUTPUT "enter untuk lanjut.."
                WAIT FOR KEYPRESS
                CLEAR SCREEN

            3 :
                DECLARE pilih_kategori : INTEGER
                CALL menu_kategori()
                SET pilih_kategori = baca_angka(1, 5)
                SET kategori = nama_kategori(pilih_kategori)
                OUTPUT "Kategori : ", kategori
                CALL header()
                CALL search_string(belajar, n, "kategori", kategori)
                OUTPUT "enter untuk lanjut.."
                WAIT FOR KEYPRESS
                CLEAR SCREEN


            4 :
                DECLARE durasi : INTEGER
                OUTPUT "Berapa durasi waktu belajar yang anda miliki (dalam menit) : "
                SET durasi = baca_angka(0, INT_MAX)
                OUTPUT "Dengan durasi waktu ", durasi, " menit anda bisa mempelajari sumber belajar : "
                CALL header()
                CALL search_durasiBelajar(belajar, n, durasi)
                OUTPUT "enter untuk lanjut.."
                WAIT FOR KEYPRESS
                CLEAR SCREEN

            5 :
                CLEAR SCREEN

            OTHERWISE : 
                CLEAR INPUT BUFFER
        END CASE
    UNTIL (pilihan == 5)
END PROCEDURE
