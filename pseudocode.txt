FUNCTION baca_angka (min : INTEGER, max : INTEGER) : INTEGER
    DECLARE angka : INTEGER
    INPUT angka

    WHILE (INPUT NOT VALID) OR (angka < min) OR (angka > max) DO 
        OUTPUT "Input salah! Masukkan angka ", min, " - ", max, " : "
        INPUT angka
    END WHILE 
    RETURN angka
END FUNCTION

FUNCTION baca_desimal (min : DOUBLE , max : DOUBLE ) : DOUBLE
    DECLARE angka : DOUBLE
    INPUT angka

    WHILE (INPUT NOT VALID) OR (angka < min) OR (angka > max) DO 
        OUTPUT "Input salah! Masukkan angka ", min, " - ", max, " : "
        INPUT angka
    END WHILE
    RETURN angka
END FUNCTION

FUNCTION nama_kategori (pilihan : INTEGER) : STRING 
    IF (pilihan == 1) THEN RETURN "buku"
    IF (pilihan == 2) THEN RETURN "catatan"
    IF (pilihan == 3) THEN RETURN "tutorial"
    IF (pilihan == 4) THEN RETURN "video_pembelajaran"
    RETURN "online_course"
END FUNCTION

PROCEDURE input (IN/OUT belajar : data_sumber, n : INTEGER) 
    DECLARE pilih : INTEGER

    SET belajar.id = n + 1

    OUTPUT "===== Menambah data sumber belajar ====="
    OUTPUT "Judul (1 kata, pakai _ untuk spasi) : "
    INPUT belajar.judul

    CALL menu_kategori()
    SET pilih = baca_angka(1, 5)
    SET belajar.kategori = nama_kategori(pilih)

    CALL menu_tingkat()
    SET pilih = baca_angka(1, 3)
    IF (pilih == 1) THEN
        SET belajar.tingkat_kesulitan = "mudah"
    ELSE IF (pilih == 2) THEN
        SET belajar.tingkat_kesulitan = "sedang"
    ELSE
        SET belajar.tingkat_kesulitan = "sulit"
    END IF

    OUTPUT "Rating (1-5) = "
    SET belajar.rating = baca_desimal(1, 5)

    OUTPUT "Durasi belajar (menit) = "
    SET belajar.waktu = baca_angka(1, INT_MAX)

    OUTPUT "Data berhasil ditambahkan (ID ", belajar.id, ")."
END PROCEDURE

PROCEDURE view_baris (belajar : data_sumber) 
    OUTPUT belajar.id, belajar.judul, belajar.kategori, belajar.tingkat_kesulitan, belajar.rating, belajar.waktu
END PROCEDURE

PROCEDURE view_arr (belajar[] : data_sumber, n : INTEGER) 
    IF (n == 0) THEN
        OUTPUT "(tidak ada data)"
    END IF
    FOR  i = 0 TO n-1 DO 
        CALL view_baris(belajar[i])
    END FOR
END PROCEDURE

PROCEDURE view_terkelompokan (tingkat : STRING, belajar[] : data_sumber, n : INTEGER) 
    DECLARE jumlah : INTEGER
    SET jumlah = 0

    FOR i = 0 TO n-1 DO 
        IF (belajar[i].tingkat_kesulitan == tingkat) THEN
            CALL view_baris(belajar[i])
            SET jumlah = jumlah + 1
        END IF
    END FOR
    IF (jumlah == 0) THEN
        OUTPUT "(tidak ada data)"
    END IF
END PROCEDURE


========================== load_save_file ========================================
PROCEDURE load (IN/OUT belajar[] : data_sumber, IN/OUT n : INTEGER) 
    OPEN FILE "data.txt" FOR READING AS out
    IF (NOT out) THEN 
        RETURN
    END IF
    WHILE (READ out INTO belajar[n].id, belajar[n].judul, belajar[n].kategori, belajar[n].tingkat_kesulitan, belajar[n].rating, belajar[n].waktu) DO 
        SET n = n + 1
    END WHILE
    CLOSE FILE out
END PROCEDURE

PROCEDURE save (belajar : data_sumber) 
    OPEN FILE "data.txt" FOR APPEND AS in

    IF (in IS OPEN) THEN 
        WRITE in FROM belajar.id, belajar.judul, belajar.kategori, belajar.tingkat_kesulitan, belajar.rating, belajar.waktu 
    ELSE
        OUTPUT "File data.txt tidak ditemukan."
    END IF
    CLOSE FILE in
END PROCEDURE

PROCEDURE bubble_sort(IN/OUT belajar[] : data_sumber, jumlah : INTEGER, properties : STRING)
    FOR i = 0 TO jumlah-1 DO
        FOR j = i+1 TO jumlah-1 DO
            IF (properties == "rating") THEN
                IF (belajar[i].rating > belajar[j].rating) THEN
                    DECLARE temp : data_sumber
                    SET temp = belajar[j]
                    SET belajar[j] = belajar[i]
                    SET belajar[i] = temp
                END IF
            END IF

            IF (properties == "waktu") THEN
                IF (belajar[i].waktu > belajar[j].waktu) THEN
                    DECLARE temp : data_sumber
                    SET temp = belajar[j]
                    SET belajar[j] = belajar[i]
                    SET belajar[i] = temp
                END IF
            END IF
        END FOR
    END FOR
END PROCEDURE

PROCEDURE search_int (belajar[] : data_sumber, n : INTEGER, properties : STRING, target : INTEGER) 
    DECLARE jumlah : INTEGER
    SET jumlah = 0
    FOR i = 0 TO n-1 DO
        IF (properties == "id") THEN
            IF (belajar[i].id == target) THEN
                CALL view_baris(belajar[i])  
                SET jumlah = jumlah + 1
            END IF
        END IF

        IF (properties == "waktu") THEN
            IF (belajar[i].waktu == target) THEN
                CALL view_baris(belajar[i])  
                SET jumlah = jumlah + 1
            END IF
        END IF
    END FOR            
    IF (jumlah == 0) THEN
        OUTPUT "(data tidak ditemukan)"
    END IF
END PROCEDURE
